import 'package:flutter/services.dart';
import 'dart:convert';
import 'package:pantry_app/game/constraints/constraint.dart';

class DictionaryService {

    final Set<String> words;
    DictionaryService._(this.words);

    static Future<DictionaryService> create() async {
        final raw = await rootBundle.loadString('assets/words.txt');
        Set<String> words = LineSplitter.split(raw)
        .map((word) => word.toLowerCase().trim())
        .where((word) => word.isNotEmpty)
        .toSet();
        return DictionaryService._(words);
    }

    bool isWordValid(String word) => words.contains(word);
    
    bool isChallengeValid(Constraint constraint){
        int wordCount = 0;
        int minSize = 100;

        for (String word in words){
            if(constraint.isAccepted(word)) wordCount++;
            if(wordCount == minSize) return true;
        }
        return false;
    }
}

import '../data/dictionary_service.dart';
import 'constraints/constraints_barrel.dart';
import 'seeded_random.dart';

class ChallengeGenerator {
  static const int maxAttempts = 20;

  String _letters = 'etaoinshrdlu';

  final SeededRandom gen;
  final DictionaryService dictionary;

  ChallengeGenerator._(this.gen, this.dictionary);

  static Future<ChallengeGenerator> create(
    DictionaryService dictionary,
  ) async {
    final rng = await SeededRandom.create();
    return ChallengeGenerator._(rng, dictionary);
  }

  late final List<Constraint Function()> pool = [
    () => ConstraintContains(getLetter()),
    () => ConstraintExcludes(getLetter()),
    () => ConstraintStartsWith(getLetter()),
    () => ConstraintEndsWith(getLetter()),
  ];

  String getLetter() {
    return _letters[gen.getRandom(0, _letters.length - 1)];
  }

  Constraint getChallenge(int difficulty) {
    switch (difficulty) {
      case 1:
        _letters = 'etaoinshrdlu';

        return _createValidChallenge(
          createChallenge,
          () => ConstraintContains('e'),
        );

      case 2:
        _letters = 'abcdefghijklmnopqrstuvwxyz';

        return _createValidChallenge(
          _createCombinedChallenge,
          () => CombinedConstraints([
            ConstraintContains('e'),
            ConstraintContains('t'),
          ]),
        );

      default:
        throw ArgumentError.value(
          difficulty,
          'difficulty',
          'Difficulty must be 1 or 2.',
        );
    }
  }

  Constraint _createValidChallenge(
    Constraint Function() buildCandidate,
    Constraint Function() buildFallback,
  ) {
    // Try at most maxAttempts randomly generated candidates.
    for (var attempt = 0; attempt < maxAttempts; attempt++) {
      final candidate = buildCandidate();

      if (dictionary.isChallengeValid(candidate)) {
        return candidate;
      }
    }

    // All random candidates failed validation.
    // Check the fallback rather than silently returning an invalid one.
    final fallback = buildFallback();

    if (!dictionary.isChallengeValid(fallback)) {
      throw StateError(
        'Could not generate a valid challenge after '
        '$maxAttempts attempts, and the fallback is also invalid.',
      );
    }

    return fallback;
  }

  Constraint _createCombinedChallenge() {
    // Create a fresh list for every combination.
    return CombinedConstraints([
      createChallenge(),
      createChallenge(),
    ]);
  }

  Constraint createChallenge() {
    final random = gen.getRandom(0, pool.length - 1);
    final picked = pool[random];

    return picked();
  }
}

import 'dart:convert';
import 'package:http/http.dart' as http;

Future<String> fetchDate() async {
  try {
    final address = Uri.parse('https://timeapi.io/api/v1/time/current/utc');

    final response = await http.get(
      address, 
      headers: {'accept': '*/*'},
    );

    if (response.statusCode == 200) {
      final Map<String, dynamic> data = jsonDecode(response.body);
      final String utcTimeString = data['utc_time'];

      // 1. Grab date part: "2026-10-10"
      // 2. Strip dashes: "20261010"
      return utcTimeString.split('T')[0].replaceAll('-', '');
    }
  } catch (e) {
    print('Failed to reach time server: $e');
  }

  // Fallback: If offline or failed, grab device time, format it, and clean it too!
  print('Using fallback device time.');
  final DateTime now = DateTime.now().toUtc();
  
  // Convert device DateTime to ISO string, grab the date part, and strip dashes
  final String deviceIsoString = now.toIso8601String();
  return deviceIsoString.split('T')[0].replaceAll('-', '');
}

import 'dart:async';

class RoundTimer {
  final void Function() onRoundEnded;

  RoundTimer(this.onRoundEnded);

  DateTime? currentTime;
  DateTime? deadline;
  Timer? _timer;

  void startTimer() {
    _timer?.cancel();

    currentTime = DateTime.now();
    deadline = currentTime!.add(
      const Duration(seconds: 10),
    );

    _timer = Timer(const Duration(seconds: 10), () {
      onRoundEnded();
    });
  }

  double? whatTimeRemains() {
    if (deadline == null) {
      return null;
    }

    final remaining = deadline!.difference(DateTime.now());

    return remaining.inMicroseconds /
        Duration.microsecondsPerSecond;
  }

  void cancelTimer() {
    _timer?.cancel();
  }
}

import 'constraint.dart';

class CombinedConstraints extends Constraint {

  final List<Constraint> constraints;

  CombinedConstraints(this.constraints);

  @override
  String get description {
    if (constraints.length != 2) return constraints.map((c) => c.description).join(' and ');

    final first = constraints[0].description;
    // Replace the prefix "Word must " with an empty string on the second rule
    final second = constraints[1].description.replaceFirst('Word must ', '');

    return '$first and $second';
  }

  @override
  bool checkConstraint(String word) {
    return constraints.every((c) => c.checkConstraint(word));
  }
}

// ---- Tuning knobs: change these to adjust the feel -------------------------
const int kTotalRounds = 7;
const double kRoundSeconds = 10;

/// How long the neutral "word captured" cue plays.
const Duration kCaptureCue = Duration(milliseconds: 450);

/// Total pause between rounds, measured from the moment the word is captured.
/// The new timer only starts after this, so the pause never costs player time.
const Duration kRoundPause = Duration(milliseconds: 1200);

/// Delay before reading the box after the timer hits zero, so the keyboard can
/// commit the final keystroke.
const Duration kAutoSubmitDelay = Duration(milliseconds: 300);
// -----------------------------------------------------------------------------

const List<String> _mockConstraints = [
  'Word must start with C',
  'Word must have 5 letters',
  'Word must contain Z',
  'Word must end with T',
  'Word must exclude E',
  'Word must start with B',
  'Word must have 7 letters',
];

enum RoundPhase { playing, transitioning }

class _RoundRecord {
  _RoundRecord(this.constraint, this.word, this.secondsLeft);
  final String constraint;
  final String word;
  final double secondsLeft;
}

class MockRoundScreen extends StatefulWidget {
  const MockRoundScreen({super.key});

  State<MockRoundScreen> createState() => _MockRoundScreenState();
}

class _MockRoundScreenState extends State<MockRoundScreen>
    with SingleTickerProviderStateMixin {
  // Runs from 1.0 (full) down to 0.0 (time up). value * kRoundSeconds is the
  // fractional number of seconds remaining, so the bar moves smoothly.
  late final AnimationController _timer;

  final TextEditingController _wordController = TextEditingController();
  final FocusNode _focusNode = FocusNode();

  RoundPhase _phase = RoundPhase.playing;
  int _round = 1;
  bool _captured = false; // drives the neutral "captured" cue
  final List<_RoundRecord> _records = [];

  bool get _isPlaying => _phase == RoundPhase.playing;

  String get _constraint =>
      _mockConstraints[(_round - 1) % _mockConstraints.length];

  @override
  void initState() {
    super.initState();
    _timer = AnimationController(
      vsync: this,
      duration: Duration(milliseconds: (kRoundSeconds * 1000).round()),
      value: 1,
    );
    _timer.addStatusListener((status) {
      if (status == AnimationStatus.dismissed) _onTimeout();
    });
    _timer.reverse();
  }

  @override
  void dispose() {
    _timer.dispose();
    _wordController.dispose();
    _focusNode.dispose();
    super.dispose();
  }

  // ---- Round flow ----------------------------------------------------------

  /// Manual submit. The guard plus the immediate phase change means a second
  /// submission (button + timeout) is impossible, whatever the timing.
  void _submit() {
    if (!_isPlaying) return;
    setState(() => _phase = RoundPhase.transitioning);
    _timer.stop();
    _finishRound(_timer.value * kRoundSeconds);
  }

  Future<void> _onTimeout() async {
    if (!_isPlaying) return;
    // Button greys out and typing is blocked immediately.
    setState(() => _phase = RoundPhase.transitioning);
    await Future.delayed(kAutoSubmitDelay);
    if (!mounted) return;
    _finishRound(0);
  }

  Future<void> _finishRound(double secondsLeft) async {
    _records.add(
      _RoundRecord(_constraint, _wordController.text.trim(), secondsLeft),
    );

    // Neutral cue, identical for every word: no right/wrong judgement.
    setState(() => _captured = true);
    await Future.delayed(kCaptureCue);
    if (!mounted) return;

    if (_round >= kTotalRounds) {
      await _showResults();
      return;
    }

    // Swap in the next round while the timer sits full and still.
    setState(() {
      _round++;
      _captured = false;
      _wordController.clear();
    });
    _timer.value = 1;

    await Future.delayed(kRoundPause - kCaptureCue);
    if (!mounted) return;

    setState(() => _phase = RoundPhase.playing);
    _timer.reverse();
  }

  void _restart() {
    setState(() {
      _records.clear();
      _round = 1;
      _captured = false;
      _wordController.clear();
      _phase = RoundPhase.playing;
    });
    _timer.value = 1;
    _timer.reverse();
  }

  // ---- Dialogs -------------------------------------------------------------

  Future<void> _confirmQuit() async {
    final wasPlaying = _isPlaying;
    if (wasPlaying) _timer.stop(); // pause the clock while the dialog is open

      context: context,
      builder: (context) => AlertDialog(
        backgroundColor: Colors.white,
        title: const Text('Quit game?'),
        content: const Text('Your progress this game will be lost.'),
        actions: [

            onPressed: () => Navigator.of(context).pop(false),
            child: const Text('Keep playing'),
          ),
          TextButton(
            onPressed: () => Navigator.of(context).pop(true),
            child: const Text('Quit'),
          ),
        ],
      ),
    );

    if (!mounted) return;
    if (quit == true) {
      Navigator.of(context).maybePop();
      return;
    }
    if (wasPlaying && _isPlaying) _timer.reverse();
  }

// Suggested location: lib/theme/app_theme.dart
import 'package:flutter/material.dart';

/// All colours live here so the whole app can be re-skinned in one place.
class AppColors {
  AppColors._();

  /// The app's signature colour.
  static const Color yellow = Color(0xFFFFD60A);

  /// A pale tint of the signature colour, used for track/background areas.
  static const Color yellowSoft = Color.fromARGB(255, 255, 206, 149);

  static const Color white = Colors.white;

  /// Near-black for text. Softer than pure #000 but still very high contrast.
  static const Color ink = Color(0xFF111111);

  static const Color inkMuted = Color.fromARGB(255, 249, 249, 249);
}

class AppTheme {
  AppTheme._();

  static ThemeData get light {
    final scheme = ColorScheme.fromSeed(
      seedColor: AppColors.yellow,
      brightness: Brightness.light,
    ).copyWith(
      primary: AppColors.yellow,
      onPrimary: AppColors.ink,
      secondary: AppColors.yellow,
      onSecondary: AppColors.ink,
      surface: AppColors.white,
      onSurface: AppColors.ink,
    );

    final base = ThemeData(useMaterial3: true, colorScheme: scheme);

    final textTheme = base.textTheme
        .apply(bodyColor: AppColors.ink, displayColor: AppColors.ink)
        .copyWith(
          // The constraint text ("Word must start with C").
          headlineMedium: base.textTheme.headlineMedium?.copyWith(
            fontSize: 32,
            fontWeight: FontWeight.w800,
            height: 1.15,
            color: AppColors.ink,
          ),
          // What the player types.
          displaySmall: base.textTheme.displaySmall?.copyWith(
            fontSize: 36,
            fontWeight: FontWeight.w700,
            color: AppColors.ink,
          ),
        );

    return base.copyWith(
      scaffoldBackgroundColor: AppColors.white,
      textTheme: textTheme,

      // The word box: solid yellow, no border until focused.
      inputDecorationTheme: InputDecorationTheme(
        filled: true,
        fillColor: AppColors.yellow,
        hintStyle: const TextStyle(color: AppColors.inkMuted),
        contentPadding:
            const EdgeInsets.symmetric(horizontal: 28, vertical: 20),
        border: OutlineInputBorder(
          borderRadius: BorderRadius.circular(24),
          borderSide: BorderSide.none,
        ),
        enabledBorder: OutlineInputBorder(
          borderRadius: BorderRadius.circular(24),
          borderSide: BorderSide.none,
        ),
        focusedBorder: OutlineInputBorder(
          borderRadius: BorderRadius.circular(24),
          borderSide: const BorderSide(color: AppColors.ink, width: 2.5),
        ),
      ),

      // Buttons: yellow with black text; clearly grey when disabled.
      elevatedButtonTheme: ElevatedButtonThemeData(
        style: ElevatedButton.styleFrom(
          backgroundColor: AppColors.yellow,
          foregroundColor: AppColors.ink,
          disabledBackgroundColor: const Color(0xFFE8E8E8),
          disabledForegroundColor: const Color(0xFFA0A0A0),
          elevation: 0,
          padding: const EdgeInsets.symmetric(horizontal: 40, vertical: 16),
          shape: const StadiumBorder(),
          textStyle: const TextStyle(fontSize: 20, fontWeight: FontWeight.w800),
        ),
      ),
    );
  }
}
